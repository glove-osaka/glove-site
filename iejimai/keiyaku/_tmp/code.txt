/**
 * 家じまい相談窓口　電子契約（受け取り・保存・メール送信の仕組み）
 * 動かす場所：家じまい専用の Google アカウントの Apps Script
 *
 * 【最初の設定】（手順書「電子契約の設定手順」も参照）
 * 1. 家じまい専用の Google アカウントで Google ドライブを開き、新しいスプレッドシートを作る（名前：家じまい 電子契約 台帳）
 * 2. 「拡張機能」→「Apps Script」を開き、最初からある文字を消して、このファイルの中身を貼り付ける
 * 2b. 左の「＋」→「HTML」で「Admin」という名前のファイルを作り、Admin.html の中身を貼り付ける
 * 3. 左の「プロジェクトの設定」（歯車）→「スクリプト プロパティ」に次の3つを登録する（合言葉は使いません）
 *      NOTIFY_TO    … 契約が結ばれたときに知らせるメールアドレス（例：家じまい専用のGmail）
 *      SIGN_URL     … お客様の契約画面のURL（https://glove-osaka.github.io/glove-site/iejimai/keiyaku/）
 *      SHEET_ID     … （単独のスクリプトで動かす場合）台帳スプレッドシートのID
 *      COMPANY_SEAL … （任意）会社の印影画像をドライブに置いたときのファイルID
 * 4. 関数の選択で「setup」を選んで「実行」→ 権限を「許可」（保存用のフォルダとシートが作られます）
 * 5. 「デプロイ」→「新しいデプロイ」→ 種類「ウェブアプリ」
 *      次のユーザーとして実行：自分　／　アクセスできるユーザー：全員
 *    → 表示された「ウェブアプリのURL」を、keiyaku/config.js の KEIYAKU_API_URL に入れる（お客様用）
 * 6. もう一度「デプロイ」→「新しいデプロイ」→ ウェブアプリ
 *      次のユーザーとして実行：自分　／　アクセスできるユーザー：自分のみ
 *    → このURLの末尾に ?admin=1 を付けたものが「管理画面」。家じまい専用のGoogleアカウントでログインしていれば、合言葉なしで開けます
 *      （keiyaku/config.js の KEIYAKU_ADMIN_URL に入れる）
 */

// 書き込みをする操作だけ順番待ち（ロック）にする。読むだけの操作は待たずに同時に動く。
var WRITE_ACTIONS = ['client.get', 'client.sign', 'admin.issue', 'admin.resend', 'admin.cancel', 'admin.password', 'admin.setlogin',
  'kanri.save', 'kanri.note', 'kanri.remove', 'kanri.report.send'];
function withLock_(action, fn) {
  if (WRITE_ACTIONS.indexOf(action) < 0) return fn();
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) return { ok: false, error: 'ただいま混み合っています。少し待ってから、もう一度押してください', busy: true };
  try { return fn(); } finally { lock.releaseLock(); }
}
function doPost(e) {
  var res;
  try {
    var req = {};
    try { req = JSON.parse(e.postData.contents || '{}'); } catch (err) { return out_({ ok: false, error: '形式が正しくありません' }); }
    var a = String(req.action || '');
    res = withLock_(a, function () {
      switch (a) {
        case 'client.get': return clientGet_(req);
        case 'client.sign': return clientSign_(req);
        case 'client.pdf': return clientPdf_(req);
        case 'admin.login': return adminLogin_(req);
        case 'admin.logout': if (/^[0-9a-f]{64}$/.test(req.session || '')) CacheService.getScriptCache().remove('s_' + req.session); return { ok: true };
        default:
          if (a.indexOf('admin.') === 0 || a.indexOf('kanri.') === 0) {
            if (!sessionOk_(req.session)) return { ok: false, error: 'ログインの有効時間が切れました。もう一度ログインしてください', relogin: true };
            if (a === 'admin.password') return adminPassChange_(req);
            return adminDispatch_(a, req);
          }
          return { ok: false, error: '不明な操作です' };
      }
    });
  } catch (err) {
    console.error(err && err.stack || err);
    res = { ok: false, error: 'うまくいきませんでした（' + String(err && err.message || err) + '）。もう一度お試しください' };
  }
  return out_(res);
}

      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
  }
  return out_({ ok: true, service: '家じまい相談窓口 電子契約' });
}

/* ---------- 管理画面から呼ばれる（家じまい専用のGoogleアカウントだけが使える） ---------- */
function adminCall(action, req) {
  var me = Session.getActiveUser().getEmail(), owner = Session.getEffectiveUser().getEmail();
  if (!me || me !== owner) return { ok: false, error: '家じまい専用の Google アカウントでログインして開いてください' };
  try {
    return withLock_(action, function () {
      if (action === 'admin.setlogin') return adminSetLogin_(req || {});
      if (action === 'admin.loginState') return { ok: true, ready: passReady_(), id: PropertiesService.getScriptProperties().getProperty('ADMIN_ID') || '' };
      return adminDispatch_(action, req || {});
    });
  } catch (err) { console.error(err && err.stack || err); return { ok: false, error: String(err.message || err) }; }
}
function adminDispatch_(action, req) {
  switch (action) {
    case 'admin.templates': return { ok: true, templates: templateList_(), fields: FIELD_DEFS };
    case 'admin.issue': return adminIssue_(req);
    case 'admin.list': return adminList_();
    case 'admin.resend': return adminResend_(req);
    case 'admin.pdf': return adminPdf_(req);
    case 'admin.cancel': return adminCancel_(req);
    default:
      if (action.indexOf('kanri.') === 0) return kanriCall_(action, req);
      return { ok: false, error: '不明な操作です' };
  }
}

/* ---------- 管理画面のログイン（ID とパスワード） ----------
 * 「プロジェクトの設定」→「スクリプト プロパティ」に、次の2つを登録します。
 *   ADMIN_ID   … ログインID（例：iejimai）
 *   ADMIN_PASS … パスワード（最初のログインのときに暗号化され、この欄は自動で消えます）
 */
var SESSION_HOURS = 6, MAX_FAIL = 5, LOCK_MIN = 15;
function hash_(pass, salt) {
  var h = salt + '|' + pass;
  for (var i = 0; i < 2000; i++) h = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, h, Utilities.Charset.UTF_8).map(function (b) { return ('0' + (b & 255).toString(16)).slice(-2); }).join('');
  return h;
}
function passReady_() {
  var P = PropertiesService.getScriptProperties(), plain = P.getProperty('ADMIN_PASS');
  if (plain) { var salt = Utilities.getUuid(); P.setProperties({ ADMIN_SALT: salt, ADMIN_PASS_HASH: hash_(plain, salt) }); P.deleteProperty('ADMIN_PASS'); }
  return !!(P.getProperty('ADMIN_ID') && P.getProperty('ADMIN_PASS_HASH'));
}
function adminLogin_(req) {
  if (!passReady_()) return { ok: false, error: 'ログインIDとパスワードがまだ登録されていません' };
  var C = CacheService.getScriptCache(), fk = 'fail', fails = +(C.get(fk) || 0);
  if (fails >= MAX_FAIL) return { ok: false, error: '何度も間違えたため、' + LOCK_MIN + '分ほどログインできません。時間をおいてお試しください' };
  var P = PropertiesService.getScriptProperties();
  var good = String(req.id || '') === P.getProperty('ADMIN_ID') && hash_(String(req.pass || ''), P.getProperty('ADMIN_SALT')) === P.getProperty('ADMIN_PASS_HASH');
  if (!good) { C.put(fk, String(fails + 1), LOCK_MIN * 60); Utilities.sleep(800); return { ok: false, error: 'ログインIDかパスワードが違います' }; }
  C.remove(fk);
  var t = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '');
  C.put('s_' + t, '1', SESSION_HOURS * 3600);
  return { ok: true, session: t, hours: SESSION_HOURS };
}
function sessionOk_(t) {
  if (!t || !/^[0-9a-f]{64}$/.test(t)) return false;
  var C = CacheService.getScriptCache(); if (!C.get('s_' + t)) return false;
  C.put('s_' + t, '1', SESSION_HOURS * 3600); return true;
}
// Google アカウント（家じまい専用）で開いた管理画面からだけ呼べる：ログインIDとパスワードの登録
function adminSetLogin_(req) {
  var id = String(req.id || '').trim(), n = String(req.pass || '');
  if (!/^[A-Za-z0-9._-]{3,30}$/.test(id)) throw new Error('ログインIDは、半角の英字と数字で3〜30文字にしてください');
  if (n.length < 10 || !/[A-Za-z]/.test(n) || !/[0-9]/.test(n)) throw new Error('パスワードは、英字と数字をまぜて10文字以上にしてください');
  var salt = Utilities.getUuid(), P = PropertiesService.getScriptProperties();
  P.setProperties({ ADMIN_ID: id, ADMIN_SALT: salt, ADMIN_PASS_HASH: hash_(n, salt) }); P.deleteProperty('ADMIN_PASS');
  CacheService.getScriptCache().remove('fail');
  return { ok: true };
}
function adminPassChange_(req) {
  var P = PropertiesService.getScriptProperties();
  if (hash_(String(req.current || ''), P.getProperty('ADMIN_SALT')) !== P.getProperty('ADMIN_PASS_HASH')) throw new Error('今のパスワードが違います');
  var n = String(req.next || '');
  if (n.length < 10 || !/[A-Za-z]/.test(n) || !/[0-9]/.test(n)) throw new Error('新しいパスワードは、英字と数字をまぜて10文字以上にしてください');
  var salt = Utilities.getUuid(); P.setProperties({ ADMIN_SALT: salt, ADMIN_PASS_HASH: hash_(n, salt) });
  return { ok: true };
}

function setup() {
  var ss = book_();

  if (sh.getLastRow() === 0) sh.appendRow(COLS);
  folder_('契約書（押印済み）');
  folder_('本人確認書類');
  Logger.log('設定が終わりました。次はウェブアプリとしてデプロイしてください。');
}
function folder_(name) {
  var root = DriveApp.getRootFolder();
  var it = root.getFoldersByName('家じまい 電子契約');
  var base = it.hasNext() ? it.next() : root.createFolder('家じまい 電子契約');
  var it2 = base.getFoldersByName(name);
  return it2.hasNext() ? it2.next() : base.createFolder(name);
}

/* ---------- 台帳 ---------- */
function book_() { var id = prop_('SHEET_ID'); return id ? SpreadsheetApp.openById(id) : SpreadsheetApp.getActiveSpreadsheet(); }
function sheet_() { var sh = book_().getSheetByName(SHEET); if (sh.getLastColumn() < COLS.length) sh.getRange(1, 1, 1, COLS.length).setValues([COLS]); return sh; }
function rows_() {
  var sh = sheet_(), v = sh.getDataRange().getValues(), out = [];
  for (var i = 1; i < v.length; i++) {
    var o = { _row: i + 1 };
    COLS.forEach(function (c, j) { o[c] = v[i][j]; });
    out.push(o);
  }
  return out;
}
function find_(key, val) { var r = rows_().filter(function (x) { return String(x[key]) === String(val); }); return r[0] || null; }
function save_(o) {
  var sh = sheet_(), line = COLS.map(function (c) { return o[c] === undefined ? '' : o[c]; });
  if (o._row) sh.getRange(o._row, 1, 1, COLS.length).setValues([line]); else sh.appendRow(line);
}
function log_(o, ev) {
  var l = []; try { l = JSON.parse(o.log || '[]'); } catch (e) { }
  l.push({ t: new Date().toISOString(), ev: ev }); o.log = JSON.stringify(l);
}

/* ---------- お客様 ---------- */
function getByToken_(t) {
  var o = t ? find_('token', t) : null;
  if (!o) throw new Error('この契約書は見つかりません。リンクをご確認いただくか、お電話ください。');
  if (o.status === '取り消し') throw new Error('この契約書は取り消されました。お電話でお問い合わせください。');
  return o;
}
function clientGet_(req) {
  var o = getByToken_(req.token);
  if (!o.viewed) { o.viewed = new Date(); if (o.status === '送付済み') o.status = '確認中'; log_(o, '開封'); save_(o); }
  var tp = TEMPLATES[o.template];
  return {
    ok: true, contract: {
      id: o.id, title: o.title, status: o.status, draft: o.draft === 'yes', party_name: o.party_name, party_company: o.party_company,
      body: fill_(tp.body, o, null), fields: JSON.parse(o.fields || '[]').map(function (k) { var d = FIELD_DEFS[k]; return { key: k, label: d.label, type: d.type, required: !!d.required }; }),
      requireId: o.require_id === 'yes', issuer: ISSUER, signed: !!o.signed
    }
  };
}
function clientSign_(req) {
  var o = getByToken_(req.token);
  if (o.status === '締結済み') throw new Error('この契約書は、すでに締結済みです。');
  if (o.draft === 'yes') throw new Error('この書面は下書きの確認用です。締結はできません。');
  var fields = JSON.parse(o.fields || '[]'), v = req.values || {};
  fields.forEach(function (k) { if (FIELD_DEFS[k].required && !String(v[k] || '').trim()) throw new Error(FIELD_DEFS[k].label + 'を入れてください'); });
  if (!req.agree) throw new Error('内容に同意のうえ、承認してください');
  if (!req.sealPng) throw new Error('押印がまだです');
  if (o.require_id === 'yes' && !req.idImage) throw new Error('本人確認書類の写真を添付してください');
  var clean = {}; fields.forEach(function (k) { clean[k] = String(v[k] || '').slice(0, 500); });
  o.values = JSON.stringify(clean);
  o.signed = new Date();
  if (req.idImage) {
    var m = String(req.idImage).match(/^data:(image\/[a-z]+);base64,(.+)$/);
    if (m) { var f = folder_('本人確認書類').createFile(Utilities.newBlob(Utilities.base64Decode(m[2]), m[1], o.id + '_本人確認.' + m[1].split('/')[1])); o.id_file_id = f.getId(); }
  }
  var text = fill_(TEMPLATES[o.template].body, o, clean);
  o.hash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, text + JSON.stringify(clean), Utilities.Charset.UTF_8)
    .map(function (b) { return ('0' + (b & 255).toString(16)).slice(-2); }).join('');
  var seal = Utilities.newBlob(Utilities.base64Decode(String(req.sealPng).split(',')[1]), 'image/png', 'seal.png');
  o.pdf_id = makePdf_(o, text, clean, seal, req.ua || '');
  o.status = '締結済み';
  log_(o, '押印・承認（締結）');
  save_(o);
  var pdf = DriveApp.getFileById(o.pdf_id).getBlob();
  MailApp.sendEmail({ to: o.party_email, subject: '【' + ISSUER.service + '】' + o.title + '（締結済み）', name: ISSUER.service,
    body: o.party_name + ' 様\n\nご契約の手続きが完了しました。押印済みの契約書をお送りします。大切に保管してください。\n\n' + ISSUER.service + '　' + ISSUER.tel, attachments: [pdf] });
  var to = prop_('NOTIFY_TO'); if (to) MailApp.sendEmail({ to: to, subject: '【締結】' + o.id + ' ' + o.title + '（' + o.party_name + '）', body: '契約が締結されました。\n\n' + o.id + '\n' + o.title + '\n' + o.party_name + '\n' + o.party_email, attachments: [pdf] });
  return pdfOut_(o);
}
function clientPdf_(req) { var o = getByToken_(req.token); if (!o.pdf_id) throw new Error('まだ締結されていません'); return pdfOut_(o); }
function pdfOut_(o) { var b = DriveApp.getFileById(o.pdf_id).getBlob(); return { ok: true, filename: o.id + '_' + o.title + '.pdf', pdf: Utilities.base64Encode(b.getBytes()) }; }

/* ---------- 契約書（PDF）をつくる ---------- */
function makePdf_(o, text, vals, sealBlob, ua) {
  var doc = DocumentApp.create(o.id + '_' + o.title), b = doc.getBody();
  b.setMarginTop(56).setMarginBottom(56).setMarginLeft(60).setMarginRight(60);
  var st = {}; st[DocumentApp.Attribute.FONT_SIZE] = 10.5; st[DocumentApp.Attribute.FONT_FAMILY] = 'Noto Serif JP'; b.setAttributes(st);
  b.appendParagraph(o.title).setHeading(DocumentApp.ParagraphHeading.HEADING1).setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  b.appendParagraph('契約番号　' + o.id).setAlignment(DocumentApp.HorizontalAlignment.RIGHT);
  text.split('\n').forEach(function (line) {
    var p = b.appendParagraph(line);
    if (/^第[0-9０-９一二三四五六七八九十]+条/.test(line)) p.setBold(true); else p.setBold(false);
  });
  b.appendParagraph('');
  b.appendParagraph('締結日　' + Utilities.formatDate(o.signed, 'Asia/Tokyo', 'yyyy年M月d日 HH:mm'));
  b.appendParagraph('');
  b.appendParagraph('【甲】').setBold(true);
  var tp = TEMPLATES[o.template] || {};
  b.appendParagraph(ISSUER.address + '\n' + ISSUER.name + (tp.issuer ? '\n買取大吉 大阪勝山南店' : '') + '\n' + ISSUER.rep + (tp.issuer ? '\n' + ISSUER.license : '')).setBold(false);
  var cs = prop_('COMPANY_SEAL'); if (cs) { try { b.appendImage(DriveApp.getFileById(cs).getBlob()).setWidth(72).setHeight(72); } catch (e) { } }
  b.appendParagraph('【乙】').setBold(true);
  var lines = []; Object.keys(vals).forEach(function (k) { if (vals[k]) lines.push(FIELD_DEFS[k].label + '：' + vals[k]); });
  b.appendParagraph(lines.join('\n')).setBold(false);
  b.appendImage(sealBlob).setWidth(64).setHeight(64);
  b.appendParagraph('');
  b.appendParagraph('本契約は電子契約により締結されました。契約書の内容の指紋（SHA-256）：' + o.hash + '\n通知先メールアドレス：' + o.party_email + '\n端末情報：' + String(ua).slice(0, 180)).setFontSize(8);
  doc.saveAndClose();
  var pdf = DriveApp.getFileById(doc.getId()).getAs('application/pdf').setName(o.id + '_' + o.title + '.pdf');
  var f = folder_('契約書（押印済み）').createFile(pdf);
  DriveApp.getFileById(doc.getId()).setTrashed(true);
  return f.getId();
}

/* ---------- 書面の差し込み ---------- */
function fill_(body, o, vals) {
  var items = []; try { items = JSON.parse(o.items || '[]'); } catch (e) { }
  var itemText = items.length ? items.map(function (it, i) { return '　' + (i + 1) + '. ' + it.name + '　数量 ' + (it.qty || 1) + '　金額 ' + Number(it.price || 0).toLocaleString('ja-JP') + '円' + (it.note ? '（' + it.note + '）' : ''); }).join('\n') +
    '\n　合計 ' + items.reduce(function (s, it) { return s + Number(it.price || 0); }, 0).toLocaleString('ja-JP') + '円' : '　（別紙のとおり）';
  var name = (vals && vals.name) || o.party_name;
  var t = body.replace(/{{party}}/g, (o.party_company ? o.party_company + '　' : '') + name)
    .replace(/{{items}}/g, itemText).replace(/{{note}}/g, o.note || 'なし')
    .replace(/{{date}}/g, Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy年M月d日'))
    .replace(/{{issuer}}/g, (TEMPLATES[o.template] && TEMPLATES[o.template].issuer) || ISSUER.name).replace(/{{tel}}/g, ISSUER.tel).replace(/{{address}}/g, ISSUER.address).replace(/{{license}}/g, ISSUER.license);
  if (o.draft === 'yes') t = '【下書き：確認用】\n' + t;
  return t;
}
function iso_(d) { return d instanceof Date ? d.toISOString() : String(d || ''); }
function esc_(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
