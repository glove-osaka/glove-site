/**
 * 家じまい相談窓口　相談フォームの受け取り（Google Apps Script・単体のスクリプト）
 *
 * 【設定の手順】（家じまい専用の Google アカウントで行う）
 * 1. script.google.com で「新しいプロジェクト」を作り、このファイルの中身を貼り付けて保存
 * 2. 下の SHEET_ID に、受付を書き込むスプレッドシートのIDを入れる
 *    NOTIFY_TO に、通知を受け取るメールアドレスを入れる
 * 3. 「デプロイ」→「新しいデプロイ」→ 種類「ウェブアプリ」
 *      次のユーザーとして実行：自分 ／ アクセスできるユーザー：全員
 * 4. 表示された URL を build2.py の CONTACT["form_url"] に入れて作り直す
 */
var SHEET_ID = '1nYJQUmGDjWd2iaPhHzI0Qy9YrT48KziMMq3xlX1clS4';
var SHEET_NAME = '相談受付';
var NOTIFY_TO = 'info@iejimai123.jp';

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
    var p = (e && e.parameter) || {};
    if (p.website) return ContentService.createTextOutput('ok'); // 迷惑投稿よけ
    var keys = Object.keys(p);
    if (sh.getLastRow() === 0) sh.appendRow(['受付日時'].concat(keys));
    var head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
    keys.forEach(function (k) { if (head.indexOf(k) < 0) { head.push(k); sh.getRange(1, head.length).setValue(k); } });
    sh.appendRow(head.map(function (h, i) { return i === 0 ? new Date() : String(p[h] || '').slice(0, 5000); }));
    var to = NOTIFY_TO;
    var body = keys.map(function (k) { return k + '：' + p[k]; }).join('\n');
    MailApp.sendEmail(to, '【家じまい相談窓口】新しいご相談が届きました', body + '\n\n受付一覧：' + ss.getUrl() + '#gid=' + sh.getSheetId());
    return ContentService.createTextOutput('ok');
  } finally {
    lock.releaseLock();
  }
}

function doGet() {
  return ContentService.createTextOutput('家じまい相談窓口 相談フォーム受付');
}

/** 最初に一度だけ実行（権限の確認と、受付シートの用意） */
function setup() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  if (!ss.getSheetByName(SHEET_NAME)) ss.insertSheet(SHEET_NAME);
  MailApp.getRemainingDailyQuota();
  Logger.log('準備ができました。次はウェブアプリとしてデプロイしてください。');
}
